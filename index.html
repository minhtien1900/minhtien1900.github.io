// Hàm loại bỏ dấu tiếng Việt function removeVietnameseTones(str) { return str .normalize('NFD') .replace(/[\u0300-\u036f]/g, '') .replace(/đ/g, 'd').replace(/Đ/g, 'D'); } // Hàm tìm kiếm và Highlight từ khóa trong văn bản function searchAndHighlight(searchText, containerElement) { if (!containerElement) return { count: 0, results: [] }; // 1. Chuẩn hóa chuỗi tìm kiếm: Xóa khoảng trắng thừa ở 2 đầu và đưa về chữ thường const cleanSearch = searchText.trim().toLowerCase(); if (!cleanSearch) return { count: 0, results: [] }; const cleanSearchNoTone = removeVietnameseTones(cleanSearch); // Lấy toàn bộ văn bản trong container const content = containerElement.innerText || containerElement.textContent; const contentLower = content.toLowerCase(); const contentNoTone = removeVietnameseTones(contentLower); let matchCount = 0; let matchIndices = []; // 2. Tìm kiếm theo cả 2 chế độ: Có dấu và Không dấu let startIndex = 0; while (startIndex < content.length) { // Tìm vị trí xuất hiện của từ khóa (so sánh chuỗi không dấu để tối ưu) const index = contentNoTone.indexOf(cleanSearchNoTone, startIndex); if (index === -1) break; matchCount++; matchIndices.push(index); startIndex = index + cleanSearchNoTone.length; } // Trả về số lượng kết quả tìm được return { count: matchCount, indices: matchIndices }; } // --- Ví dụ tích hợp vào sự kiện ô tìm kiếm (Input Event) --- const searchInput = document.querySelector('#search-input'); // Thay ID tương ứng với UI của bạn const contentContainer = document.querySelector('#reader-content'); // Container chứa văn bản sách searchInput.addEventListener('input', (e) => { const keyword = e.target.value; const result = searchAndHighlight(keyword, contentContainer); // Cập nhật UI hiển thị số kết quả (ví dụ: 3/5 hoặc 0/0) const resultCounter = document.querySelector('#result-counter'); if (resultCounter) { resultCounter.textContent = `${result.count > 0 ? 1 : 0}/${result.count}`; } });
