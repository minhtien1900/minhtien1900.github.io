// ========================================== // SMART SEARCH ENGINE - VERSION 27.0 // Features: Accent-insensitive, case-insensitive, auto-scroll, real-time highlight // ========================================== const SmartSearch = { version: "V27.0", currentIndex: -1, matches: [], // Hàm loại bỏ dấu tiếng Việt removeAccents(str) { if (!str) return ""; return str .normalize("NFD") .replace(/[\u0300-\u036f]/g, "") .replace(/đ/g, "d") .replace(/Đ/g, "D") .toLowerCase(); }, // Xóa tất cả highlight cũ clearHighlights() { const highlights = document.querySelectorAll(".search-highlight"); highlights.forEach((el) => { const parent = el.parentNode; parent.replaceChild(document.createTextNode(el.textContent), el); parent.normalize(); // Gộp lại các text node bị chia nhỏ }); this.matches = []; this.currentIndex = -1; this.updateUI(); }, // Thực hiện tìm kiếm và highlight trong container search(keyword, containerId = "content-area") { this.clearHighlights(); const trimmed = keyword.trim(); if (!trimmed) return; const container = document.getElementById(containerId); if (!container) return; const normalizedKeyword = this.removeAccents(trimmed); const walker = document.createTreeWalker( container, NodeFilter.SHOW_TEXT, null, false ); const nodesToProcess = []; let node; while ((node = walker.nextNode())) { // Bỏ qua các thẻ script, style hoặc các thẻ chứa UI tìm kiếm const parentTag = node.parentNode.tagName.toLowerCase(); if (["script", "style", "input", "textarea"].includes(parentTag)) continue; const normalizedText = this.removeAccents(node.nodeValue); if (normalizedText.includes(normalizedKeyword)) { nodesToProcess.push(node); } } // Tiến hành highlight từng text node nodesToProcess.forEach((textNode) => { this.highlightNode(textNode, trimmed); }); this.matches = Array.from(document.querySelectorAll(".search-highlight")); if (this.matches.length > 0) { this.currentIndex = 0; this.focusMatch(this.currentIndex); } this.updateUI(); }, // Highlight các đoạn khớp trong textNode highlightNode(node, keyword) { const parent = node.parentNode; const text = node.nodeValue; const normText = this.removeAccents(text); const normKeyword = this.removeAccents(keyword); const fragment = document.createDocumentFragment(); let lastIndex = 0; let matchIndex = normText.indexOf(normKeyword, lastIndex); while (matchIndex !== -1) { // Thêm phần văn bản trước từ khóa if (matchIndex > lastIndex) { fragment.appendChild( document.createTextNode(text.substring(lastIndex, matchIndex)) ); } // Tạo thẻ span highlight cho từ khóa const matchText = text.substring(matchIndex, matchIndex + keyword.length); const span = document.createElement("span"); span.className = "search-highlight"; span.textContent = matchText; fragment.appendChild(span); lastIndex = matchIndex + keyword.length; matchIndex = normText.indexOf(normKeyword, lastIndex); } // Thêm phần văn bản còn lại if (lastIndex < text.length) { fragment.appendChild(document.createTextNode(text.substring(lastIndex))); } parent.replaceChild(fragment, node); }, // Focus và cuộn tới vị trí kết quả đang chọn focusMatch(index) { if (this.matches.length === 0 || index < 0 || index >= this.matches.length) return; this.matches.forEach((el, i) => { if (i === index) { el.classList.add("active-highlight"); el.scrollIntoView({ behavior: "smooth", block: "center" }); } else { el.classList.remove("active-highlight"); } }); }, // Chuyển kết quả tiếp theo next() { if (this.matches.length === 0) return; this.currentIndex = (this.currentIndex + 1) % this.matches.length; this.focusMatch(this.currentIndex); this.updateUI(); }, // Quay lại kết quả trước prev() { if (this.matches.length === 0) return; this.currentIndex = (this.currentIndex - 1 + this.matches.length) % this.matches.length; this.focusMatch(this.currentIndex); this.updateUI(); }, // Cập nhật đếm kết quả trên UI (ví dụ: 1/5) updateUI() { const countEl = document.getElementById("search-count"); if (!countEl) return; if (this.matches.length === 0) { countEl.textContent = "0/0"; } else { countEl.textContent = `${this.currentIndex + 1}/${this.matches.length}`; } } };
