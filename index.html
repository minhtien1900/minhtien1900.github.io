// 1. Hàm Debounce chống giật lag khi gõ nhanh function debounce(func, delay = 300) { let timeoutId; return (...args) => { clearTimeout(timeoutId); timeoutId = setTimeout(() => func.apply(this, args), delay); }; } // 2. Hàm chuẩn hóa bỏ dấu tiếng Việt function removeVietnameseTones(str) { return str .normalize('NFD') .replace(/[\u0300-\u036f]/g, '') .replace(/đ/g, 'd').replace(/Đ/g, 'D'); } // 3. Hàm tìm kiếm siêu tốc (Tối ưu bộ nhớ & CPU) function FastSearch(searchText, contentText) { // Xóa khoảng trắng thừa 2 đầu const cleanSearch = searchText.trim(); if (!cleanSearch) return { count: 0, indices: [] }; // Chuẩn hóa không dấu & chữ thường const searchNormalized = removeVietnameseTones(cleanSearch.toLowerCase()); const contentNormalized = removeVietnameseTones(contentText.toLowerCase()); // Escaping các ký tự đặc biệt Regex để tránh lỗi crash const escapedSearch = searchNormalized.replace(/[.*+?^${}()|[\]\\]/g, '\\$&'); // Dùng Regex toàn cục để tìm cực nhanh const regex = new RegExp(escapedSearch, 'gi'); const matches = []; let match; while ((match = regex.exec(contentNormalized)) !== null) { matches.push(match.index); // Tránh vòng lặp vô tận nếu regex khớp chuỗi rỗng if (regex.lastIndex === match.index) regex.lastIndex++; } return { count: matches.length, indices: matches }; } // --- TÍCH HỢP VÀO XỬ LÝ SỰ KIỆN (EVENT LISTENER) --- const searchInput = document.querySelector('#search-input'); // Ô tìm kiếm const contentContainer = document.querySelector('#reader-content'); // Container chứa sách const counterDisplay = document.querySelector('#result-counter'); // Ô hiển thị 0/0 // Cache lại nội dung text của sách để không phải đọc lại DOM nhiều lần gây lag let cachedContentText = ""; if (contentContainer) { cachedContentText = contentContainer.innerText || contentContainer.textContent; } // Hàm thực thi tìm kiếm đã bọc Debounce const handleSearch = debounce((event) => { const keyword = event.target.value; // Nếu từ khóa quá ngắn (< 2 ký tự), bỏ qua để đỡ tốn tài nguyên if (keyword.trim().length < 2) { if (counterDisplay) counterDisplay.textContent = "0/0"; return; } // Thực hiện tìm kiếm siêu tốc const result = FastSearch(keyword, cachedContentText); // Cập nhật giao diện đếm số kết quả if (counterDisplay) { counterDisplay.textContent = result.count > 0 ? `1/${result.count}` : "0/0"; } // TODO: Bổ sung logic scroll tới vị trí match đầu tiên nếu cần }, 300); // Trễ 300ms sau khi ngừng gõ mới chạy // Gán sự kiện input searchInput.addEventListener('input', handleSearch);
